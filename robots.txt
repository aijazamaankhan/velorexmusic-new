# =============================================================================
# Velorex Music — https://velorexmusic.com
#
# Policy: the catalogue is fully crawlable. Everything that is transactional,
# personal, or infinite (filter permutations) is kept out of the index.
#
# Note on Disallow vs noindex: Disallow stops the crawl, it does NOT remove a
# URL that is already indexed. Pages that must actually leave the index carry
# a `noindex` robots meta tag from src/js/seo.js instead — a page has to be
# crawlable for Google to see that tag.
# =============================================================================

User-agent: *

# --- Admin panel -------------------------------------------------------------
# The admin page is deliberately NOT named here. Listing it in a public
# robots.txt advertised the one URL its unconventional name exists to hide.
# It carries <meta name="robots" content="noindex, nofollow, noarchive">,
# which keeps it out of the index without announcing where it is.
Disallow: /old%20admin/

# --- Backend + tooling -------------------------------------------------------
# The public, read-only endpoints the storefront calls to BUILD its pages are
# allowed: Googlebot renders the SPA and fetches exactly these, and with all of
# /api/ blocked, the homepage's product grids, combos and blog cards rendered
# empty for Google. The responses carry "X-Robots-Tag: noindex" (api/.htaccess),
# so the JSON itself never becomes a search result. Longest match wins, so
# these Allow lines beat the Disallow below.
Allow: /api/products.php
Allow: /api/product.php
Allow: /api/categories.php
Allow: /api/combos.php
Allow: /api/blog.php
Allow: /api/music-history.php
Allow: /api/settings.php
Allow: /api/collections.php
Disallow: /api/
Disallow: /scripts/
Disallow: /node_modules/
Disallow: /test-results/
Disallow: /seo-render.php

# --- Transactional / personal views ------------------------------------------
# /cart, /checkout, /profile, /login, /signup, /forgot and /track-order.html are
# NOT disallowed, on purpose. Each is served with "noindex" in its own HTML
# (seo-render.php _route=private; track-order.html's meta tag). A Disallow
# would stop Google reading that noindex, and a blocked URL can still be
# indexed bare from links — so they stay crawlable and say "noindex" themselves.

# --- Crawl-budget control ----------------------------------------------------
# Faceted filtering can generate effectively unlimited URL permutations from
# the same products. The curated /vinyl-records/hindi style paths ARE indexable;
# these query-string forms are the uncontrolled ones.
Disallow: /*?search=
Disallow: /*?sort=
Disallow: /*&search=
Disallow: /*&sort=
Disallow: /*?people=
Disallow: /*&people=
Disallow: /*?label=
Disallow: /*&label=

# --- Private one-time links --------------------------------------------------
# Cart-recovery and coupon links from emails. They carry a token or a personal
# code and render nothing a searcher could use.
Disallow: /*?recover=
Disallow: /*?coupon=

# --- Assets ------------------------------------------------------------------
# Explicitly allowed: Google must fetch CSS/JS to render the SPA, and blocking
# them causes "page cannot be rendered" errors in Search Console. Blocking
# product images would also forfeit Google Images traffic, which for a store
# selling physical records with distinctive cover art is real volume.
Allow: /src/
Allow: /uploads/
Allow: /*.css$
Allow: /*.js$
Allow: /*.jpg$
Allow: /*.jpeg$
Allow: /*.png$
Allow: /*.webp$

# --- Aggressive SEO scrapers -------------------------------------------------
# These crawl hard, cost real bandwidth on shared hosting, and send no traffic.
User-agent: AhrefsBot
Disallow: /

User-agent: SemrushBot
Disallow: /

User-agent: MJ12bot
Disallow: /

User-agent: DotBot
Disallow: /
